/* ================================================================ ESP32 WI-FI ACTIVITY SENSOR - VERSION 1.0 ================================================================ Features ---------------------------------------------------------------- - ESP32 Wi-Fi connection - RSSI measurement - Automatic baseline calibration - Exponential moving average filter - Activity calculation - Presence/activity state - Adaptive baseline - Confidence estimation - Web dashboard hosted by ESP32 - Live signal graph - Radar-style activity visualization - Activity history - REST-style JSON endpoint - Automatic Wi-Fi reconnect - Uptime monitoring - Sensor calibration - Adjustable sensitivity - Mobile responsive interface Hardware ---------------------------------------------------------------- ESP32 development board Network ---------------------------------------------------------------- Your PLDT Home Fiber Wi-Fi IMPORTANT ---------------------------------------------------------------- This is an experimental Wi-Fi sensing project. RSSI changes can be caused by: - People - Moving objects - Doors - Furniture - Other Wi-Fi devices - Interference - Router activity Therefore this system should NOT be treated as a reliable security, surveillance, or person-identification system. ================================================================ */ #include #include // --------------------------------------------------------------- // USER CONFIGURATION // --------------------------------------------------------------- const char* WIFI_SSID = "YOUR_PLDT_WIFI"; const char* WIFI_PASSWORD = "YOUR_WIFI_PASSWORD"; // Device name const char* DEVICE_NAME = "ESP32-WIFI-SENSOR-01"; // Web server WebServer server(80); // --------------------------------------------------------------- // SENSOR SETTINGS // --------------------------------------------------------------- // How often RSSI is sampled const unsigned long SENSOR_INTERVAL = 250; // Dashboard update interval const unsigned long DASHBOARD_INTERVAL = 500; // Calibration duration const unsigned long CALIBRATION_TIME = 10000; // Number of samples kept in history const int HISTORY_SIZE = 120; // Default detection threshold float detectionThreshold = 3.5; // Stronger filtering = smoother but slower response float filterStrength = 0.18; // Baseline adaptation float baselineAdaptation = 0.008; // --------------------------------------------------------------- // SENSOR VARIABLES // --------------------------------------------------------------- int rawRSSI = 0; float filteredRSSI = 0.0; float baselineRSSI = 0.0; float signalDifference = 0.0; float activityLevel = 0.0; float confidence = 0.0; bool activityDetected = false; bool calibrated = false; bool sensorReady = false; // --------------------------------------------------------------- // TIMING // --------------------------------------------------------------- unsigned long bootTime = 0; unsigned long lastSensorRead = 0; unsigned long lastDashboardUpdate = 0; unsigned long calibrationStart = 0; unsigned long lastWiFiCheck = 0; // --------------------------------------------------------------- // CALIBRATION VARIABLES // --------------------------------------------------------------- float calibrationSum = 0.0; int calibrationSamples = 0; // --------------------------------------------------------------- // HISTORY // --------------------------------------------------------------- float activityHistory[HISTORY_SIZE]; float rssiHistory[HISTORY_SIZE]; unsigned long timeHistory[HISTORY_SIZE]; int historyIndex = 0; int historyCount = 0; // --------------------------------------------------------------- // EVENT COUNTERS // --------------------------------------------------------------- unsigned long totalSamples = 0; unsigned long detectedEvents = 0; unsigned long wifiReconnects = 0; // --------------------------------------------------------------- // STATUS // --------------------------------------------------------------- String lastEvent = "System started"; unsigned long lastEventTime = 0; // --------------------------------------------------------------- // UTILITY FUNCTIONS // --------------------------------------------------------------- float absoluteFloat(float value) { if (value < 0) { return -value; } return value; } // --------------------------------------------------------------- // WIFI STATUS // --------------------------------------------------------------- String wifiStatusText() { if (WiFi.status() == WL_CONNECTED) { return "CONNECTED"; } return "DISCONNECTED"; } // --------------------------------------------------------------- // WIFI RSSI // --------------------------------------------------------------- int readRSSI() { if (WiFi.status() != WL_CONNECTED) { return -100; } return WiFi.RSSI(); } // --------------------------------------------------------------- // ADD HISTORY SAMPLE // --------------------------------------------------------------- void addHistory(float activity, float rssi) { activityHistory[historyIndex] = activity; rssiHistory[historyIndex] = rssi; timeHistory[historyIndex] = millis() - bootTime; historyIndex++; if (historyIndex >= HISTORY_SIZE) { historyIndex = 0; } if (historyCount < HISTORY_SIZE) { historyCount++; } } // --------------------------------------------------------------- // CALIBRATION // --------------------------------------------------------------- void startCalibration() { calibrated = false; sensorReady = false; calibrationStart = millis(); calibrationSum = 0.0; calibrationSamples = 0; activityLevel = 0.0; activityDetected = false; lastEvent = "Calibration started"; lastEventTime = millis(); Serial.println(); Serial.println("--------------------------------"); Serial.println("CALIBRATION STARTED"); Serial.println("Keep the environment as still"); Serial.println("as possible for 10 seconds."); Serial.println("--------------------------------"); } // --------------------------------------------------------------- // PROCESS CALIBRATION // --------------------------------------------------------------- void processCalibration() { if (calibrated) { return; } if (WiFi.status() != WL_CONNECTED) { return; } int rssi = readRSSI(); if (rssi <= -100) { return; } calibrationSum += rssi; calibrationSamples++; unsigned long elapsed = millis() - calibrationStart; if (elapsed >= CALIBRATION_TIME) { if (calibrationSamples > 0) { baselineRSSI = calibrationSum / calibrationSamples; filteredRSSI = baselineRSSI; calibrated = true; sensorReady = true; lastEvent = "Calibration complete"; lastEventTime = millis(); Serial.println(); Serial.println("--------------------------------"); Serial.println("CALIBRATION COMPLETE"); Serial.print("Baseline RSSI: "); Serial.println(baselineRSSI); Serial.print("Samples: "); Serial.println(calibrationSamples); Serial.println("--------------------------------"); } } } // --------------------------------------------------------------- // RSSI FILTER // --------------------------------------------------------------- void updateFilteredRSSI(float newRSSI) { if (filteredRSSI == 0) { filteredRSSI = newRSSI; return; } filteredRSSI = filteredRSSI * (1.0 - filterStrength) + newRSSI * filterStrength; } // --------------------------------------------------------------- // BASELINE ADAPTATION // --------------------------------------------------------------- void updateBaseline() { if (!calibrated) { return; } /* Only adapt baseline when the environment appears stable. This prevents the baseline from immediately following a movement event. */ if (activityLevel < detectionThreshold * 0.5) { baselineRSSI = baselineRSSI * (1.0 - baselineAdaptation) + filteredRSSI * baselineAdaptation; } } // --------------------------------------------------------------- // ACTIVITY CALCULATION // --------------------------------------------------------------- void calculateActivity() { signalDifference = absoluteFloat( filteredRSSI - baselineRSSI ); /* Smooth activity value */ activityLevel = activityLevel * 0.70 + signalDifference * 0.30; /* Convert activity to confidence. */ float normalized = activityLevel / (detectionThreshold * 3.0); if (normalized > 1.0) { normalized = 1.0; } if (normalized < 0.0) { normalized = 0.0; } confidence = normalized * 100.0; bool previousState = activityDetected; /* Detection hysteresis. This prevents rapid ON/OFF switching. */ if (!activityDetected) { if (activityLevel >= detectionThreshold) { activityDetected = true; } } else { if (activityLevel < detectionThreshold * 0.65) { activityDetected = false; } } /* Detect new event. */ if (activityDetected && !previousState) { detectedEvents++; lastEvent = "Activity detected"; lastEventTime = millis(); Serial.println( "[EVENT] Activity detected" ); } if (!activityDetected && previousState) { lastEvent = "Environment stable"; lastEventTime = millis(); Serial.println( "[EVENT] Environment stable" ); } } // --------------------------------------------------------------- // SENSOR LOOP // --------------------------------------------------------------- void updateSensor() { if (millis() - lastSensorRead < SENSOR_INTERVAL) { return; } lastSensorRead = millis(); if (WiFi.status() != WL_CONNECTED) { return; } rawRSSI = readRSSI(); if (rawRSSI <= -100) { return; } totalSamples++; /* Calibration phase */ if (!calibrated) { processCalibration(); return; } /* Normal sensor processing */ updateFilteredRSSI( rawRSSI ); calculateActivity(); updateBaseline(); addHistory( activityLevel, filteredRSSI ); } // --------------------------------------------------------------- // WIFI RECONNECTION // --------------------------------------------------------------- void checkWiFi() { if (millis() - lastWiFiCheck < 5000) { return; } lastWiFiCheck = millis(); if (WiFi.status() == WL_CONNECTED) { return; } Serial.println( "Wi-Fi disconnected." ); Serial.println( "Attempting reconnection..." ); wifiReconnects++; WiFi.disconnect(); delay(100); WiFi.begin( WIFI_SSID, WIFI_PASSWORD ); } // --------------------------------------------------------------- // FORMAT UPTIME // --------------------------------------------------------------- String formatUptime( unsigned long seconds ) { unsigned long days = seconds / 86400; seconds %= 86400; unsigned long hours = seconds / 3600; seconds %= 3600; unsigned long minutes = seconds / 60; seconds %= 60; String result = ""; result += String(days); result += "d "; result += String(hours); result += "h "; result += String(minutes); result += "m "; result += String(seconds); result += "s"; return result; } // --------------------------------------------------------------- // JSON ESCAPE // --------------------------------------------------------------- String jsonEscape( String text ) { text.replace( "\\", "\\\\" ); text.replace( "\"", "\\\"" ); return text; } // --------------------------------------------------------------- // CREATE SENSOR JSON // --------------------------------------------------------------- String createSensorJSON() { String json = "{"; json += "\"device\":\""; json += jsonEscape( DEVICE_NAME ); json += "\","; json += "\"wifi\":\""; json += wifiStatusText(); json += "\","; json += "\"rssi\":"; json += String(rawRSSI); json += ","; json += "\"filteredRSSI\":"; json += String( filteredRSSI, 2 ); json += ","; json += "\"baselineRSSI\":"; json += String( baselineRSSI, 2 ); json += ","; json += "\"difference\":"; json += String( signalDifference, 2 ); json += ","; json += "\"activity\":"; json += String( activityLevel, 2 ); json += ","; json += "\"confidence\":"; json += String( confidence, 1 ); json += ","; json += "\"detected\":"; json += activityDetected ? "true" : "false"; json += ","; json += "\"calibrated\":"; json += calibrated ? "true" : "false"; json += ","; json += "\"samples\":"; json += String( totalSamples ); json += ","; json += "\"events\":"; json += String( detectedEvents ); json += ","; json += "\"reconnects\":"; json += String( wifiReconnects ); json += ","; json += "\"uptime\":"; json += String( (millis() - bootTime) / 1000 ); json += ","; json += "\"lastEvent\":\""; json += jsonEscape( lastEvent ); json += "\""; json += "}"; return json; } // --------------------------------------------------------------- // WEB API // --------------------------------------------------------------- void handleAPI() { server.sendHeader( "Access-Control-Allow-Origin", "*" ); server.send( 200, "application/json", createSensorJSON() ); } // --------------------------------------------------------------- // RESET CALIBRATION API // --------------------------------------------------------------- void handleCalibration() { startCalibration(); server.send( 200, "application/json", "{\"status\":\"calibration_started\"}" ); } // --------------------------------------------------------------- // SET SENSITIVITY API // --------------------------------------------------------------- void handleSensitivity() { if (!server.hasArg("value")) { server.send( 400, "application/json", "{\"error\":\"missing value\"}" ); return; } float value = server.arg("value") .toFloat(); /* Keep safe limits. */ if (value < 0.5) { value = 0.5; } if (value > 20.0) { value = 20.0; } detectionThreshold = value; String response = "{\"threshold\":" + String( detectionThreshold, 2 ) + "}"; server.send( 200, "application/json", response ); } // --------------------------------------------------------------- // HISTORY API // --------------------------------------------------------------- void handleHistory() { String json = "["; int start = historyCount < HISTORY_SIZE ? 0 : historyIndex; for ( int i = 0; i < historyCount; i++ ) { int index = (start + i) % HISTORY_SIZE; if (i > 0) { json += ","; } json += "{"; json += "\"activity\":"; json += String( activityHistory[index], 2 ); json += ","; json += "\"rssi\":"; json += String( rssiHistory[index], 2 ); json += ","; json += "\"time\":"; json += String( timeHistory[index] / 1000 ); json += "}"; } json += "]"; server.send( 200, "application/json", json ); } // --------------------------------------------------------------- // HTML DASHBOARD // --------------------------------------------------------------- const char MAIN_PAGE[] PROGMEM = R"HTML( ESP32 Wi-Fi Sensor

} // --------------------------------------------------------------- // 404 // --------------------------------------------------------------- void handleNotFound() { String message = "ESP32 Wi-Fi Sensor\n\n"; message += "Available endpoints:\n"; message += "/\n"; message += "/api\n"; message += "/history\n"; message += "/calibrate\n"; message += "/sensitivity?value=3.5\n"; server.send( 404, "text/plain", message ); } // --------------------------------------------------------------- // SETUP // --------------------------------------------------------------- void setup() { Serial.begin( 115200 ); delay(1000); bootTime = millis(); /* Initialize history */ for ( int i = 0; i < HISTORY_SIZE; i++ ) { activityHistory[i] = 0; rssiHistory[i] = -100; timeHistory[i] = 0; } Serial.println(); Serial.println( "==========================================" ); Serial.println( " ESP32 WI-FI ACTIVITY SENSOR" ); Serial.println( " VERSION 1.0" ); Serial.println( "==========================================" ); Serial.println(); /* Wi-Fi configuration */ WiFi.mode( WIFI_STA ); WiFi.setAutoReconnect( true ); WiFi.persistent( false ); Serial.print( "Connecting to: " ); Serial.println( WIFI_SSID ); WiFi.begin( WIFI_SSID, WIFI_PASSWORD ); int attempts = 0; while ( WiFi.status() != WL_CONNECTED && attempts < 60 ) { delay(500); Serial.print( "." ); attempts++; } Serial.println(); if ( WiFi.status() != WL_CONNECTED ) { Serial.println( "Wi-Fi connection failed." ); Serial.println( "Check SSID and password." ); } else { Serial.println( "Wi-Fi connected." ); Serial.print( "IP address: " ); Serial.println( WiFi.localIP() ); Serial.print( "Initial RSSI: " ); Serial.println( WiFi.RSSI() ); } /* HTTP routes */ server.on( "/", HTTP_GET, handleRoot ); server.on( "/api", HTTP_GET, handleAPI ); server.on( "/history", HTTP_GET, handleHistory ); server.on( "/calibrate", HTTP_GET, handleCalibration ); server.on( "/sensitivity", HTTP_GET, handleSensitivity ); server.onNotFound( handleNotFound ); /* Start server */ server.begin(); Serial.println( "HTTP server started." ); /* Start calibration */ startCalibration(); Serial.println(); Serial.println( "Open the ESP32 IP address" ); Serial.println( "from a device connected" ); Serial.println( "to the same Wi-Fi network." ); Serial.println(); Serial.println( "Example:" ); Serial.println( "http://192.168.1.100" ); } // --------------------------------------------------------------- // MAIN LOOP // --------------------------------------------------------------- void loop() { /* Handle browser requests */ server.handleClient(); /* Keep Wi-Fi alive */ checkWiFi(); /* Sensor processing */ updateSensor(); /* Calibration */ processCalibration(); /* Dashboard timing (HTTP polling is handled by browser) */ if ( millis() - lastDashboardUpdate > DASHBOARD_INTERVAL ) { lastDashboardUpdate = millis(); } /* Small cooperative delay. Prevents the loop from consuming unnecessary CPU continuously. */ delay(2); }
